# smart-electricity-dashboard
# Simple Raspberry Pi Energy Monitor

voltage = 230.0      # Volts
current = 2.0        # Amps

# Calculate power
power = voltage * current

# Display results
print("Smart Electricity Monitor")
print("-------------------------")
print("Voltage :", voltage, "V")
print("Current :", current, "A")
print("Power   :", power, "W")
